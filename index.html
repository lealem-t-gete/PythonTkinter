What is tkinter?

Python's built-in GUI library

tkinter comes installed with Python — no pip install needed. It lets you build real desktop windows with buttons, inputs, dropdowns, and more.

🐍

Built into Python

tkinter ships with every Python installation. Just import tkinter as tk and you're ready — no packages to install.

🪟

Cross-platform

Write once, run on Windows, macOS, and Linux. Your app's window adapts to the native look of each operating system automatically.

🧩

Widget-based

Everything is a widget — labels, buttons, text boxes, dropdowns. You place them on a window (called the "root") and arrange them using pack, grid, or place.

⚡

Event-driven

tkinter uses an event loop — root.mainloop(). The window waits for the user to interact, then runs your functions in response.

tk.Label

Displays text or an image on screen. Labels are read-only — users can't edit them. Use them for titles, field names, and instructions.

  • text=The string to display
  • font=("Arial", 12) — family and size
  • fg= / bg=Foreground / background colour
  • width=Width in characters
  • anchor=Text alignment: "w", "center", "e"
import tkinter as tk

root = tk.Tk()
root.geometry("300x150")

# Simple label — .pack() centres it
tk.Label(root, text="Hello, class!",
       font=("Arial", 16),
       fg="#2ecc71").pack(pady=20)

# Positioned label
tk.Label(root, text="First Name").place(x=20, y=80)

root.mainloop()

tk.Entry

A single-line text input box. Save it in a variable so you can read what the user typed using .get().

  • width=Width in characters (default 20)
  • show=Use "*" for password fields
  • state="normal", "disabled", "readonly"
  • .get()Returns current text as a string
  • .delete(0,END)Clears the field
import tkinter as tk

root = tk.Tk()
root.geometry("300x150")

tk.Label(root, text="First Name").place(x=20, y=20)

# Save entry in a variable to read later
entry_fname = tk.Entry(root, width=20)
entry_fname.place(x=20, y=45)

def read_name():
    name = entry_fname.get()
    print(f"Hello {name}!")

tk.Button(root, text="Read", command=read_name).place(x=20, y=80)
root.mainloop()

tk.Button

A clickable button. Use command= to point to a function — but don't use brackets, or it runs immediately!

  • text=Label on the button
  • command=Function to call (no brackets!)
  • width=Width in characters
  • bg= / fg=Background / text colour
  • state="normal" or "disabled"
import tkinter as tk

root = tk.Tk()
root.geometry("300x120")

def say_hello():
    print("Button clicked!")

# command= takes the function NAME, no brackets
tk.Button(root,
        text="Click Me",
        width=20,
        command=say_hello).pack(pady=30)

root.mainloop()

tk.Spinbox

A number picker with up/down arrows. Set the minimum with from_= (underscore because from is a Python keyword) and maximum with to=.

  • from_=Minimum value (note the underscore)
  • to=Maximum value
  • width=Width in characters
  • increment=Step size (default 1)
  • .get()Returns current value as a string
import tkinter as tk

root = tk.Tk()
root.geometry("300x120")

tk.Label(root, text="Age").place(x=20, y=20)

# from_= has underscore — "from" is a Python keyword
spin_age = tk.Spinbox(root, from_=16, to=99, width=12)
spin_age.place(x=20, y=45)

def show_age():
    print(spin_age.get())

tk.Button(root, text="Print age", command=show_age).place(x=20, y=80)
root.mainloop()

ttk.Combobox

A dropdown menu from the ttk module (themed widgets). Pass a Python list to values= to populate the options.

  • values=A list of options to show
  • width=Width in characters
  • state="readonly" prevents typing
  • .get()Returns the selected option
  • .set("x")Sets the default value
import tkinter as tk
from tkinter import ttk # ttk has nicer widgets

root = tk.Tk()
root.geometry("300x120")

tk.Label(root, text="Title").place(x=20, y=20)

titles = ["Mr", "Ms", "Mrs", "Dr", "Prof"]
combo = ttk.Combobox(root, values=titles, width=12)
combo.place(x=20, y=45)

tk.Button(root, text="Print",
        command=lambda: print(combo.get())).place(x=20, y=80)
root.mainloop()

tk.Checkbutton

A tick box. It must be linked to a tk.BooleanVar() — the variable stores True when ticked, False when not.

  • text=Label next to the tick box
  • variable=A BooleanVar() to store the state
  • onvalue= / offvalue=Custom values (default True/False)
  • .get()Returns True or False
  • .set(True)Ticks the box programmatically
import tkinter as tk

root = tk.Tk()
root.geometry("300x140")

# BooleanVar holds the tick state
var_reg = tk.BooleanVar()
var_terms = tk.BooleanVar()

tk.Checkbutton(root, text="Currently Registered",
             variable=var_reg).place(x=20, y=30)

tk.Checkbutton(root, text="Accept terms",
             variable=var_terms).place(x=20, y=60)

tk.Button(root, text="Check",
        command=lambda: print(var_reg.get(), var_terms.get())
        ).place(x=20, y=100)
root.mainloop()

tk.LabelFrame

A bordered box with a title, used to group related widgets. Widgets placed inside it use the frame as their parent instead of root.

  • text=Title shown in the frame border
  • padx= / pady=Inner padding in pixels
  • width= / height=Size in pixels (used with .place())
  • relief="groove", "ridge", "flat", "sunken"
import tkinter as tk

root = tk.Tk()
root.geometry("350x200")

# LabelFrame — place at a position on root
frame = tk.LabelFrame(root, text="User Information",
                    padx=5, pady=5)
frame.place(x=10, y=10, width=330, height=120)

# Widgets inside use frame as parent, not root
tk.Label(frame, text="First Name").place(x=10, y=5)
entry = tk.Entry(frame, width=15)
entry.place(x=10, y=30)

root.mainloop()

Step 1 — Create the window

Every tkinter program has the same skeleton: import, create the window, set its size and title, then call mainloop() to keep it open.

import tkinter as tk
from tkinter import ttk

root = tk.Tk() # create the window
root.geometry("500x460") # width x height
root.title("Data Entry Form") # title bar text

# Always the last line — keeps the window open
root.mainloop()

Step 2 — Add labels with .place()

Labels display text. Use .place(x=, y=) to put them at exact pixel coordinates — x goes right, y goes down.

import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.geometry("500x460")
root.title("Data Entry Form")

# Labels — .place(x, y) sets exact pixel position
tk.Label(root, text="First Name").place(x=60, y=30)
tk.Label(root, text="Last Name").place(x=200, y=30)
tk.Label(root, text="Title").place(x=360, y=30)
tk.Label(root, text="Age").place(x=60, y=90)
tk.Label(root, text="Nationality").place(x=200, y=90)

root.mainloop()

A LabelFrame draws a titled border around a group of widgets. Crucially, widgets placed inside it use the frame as their parent, not root.

Complete form — all elements together 🎉

The full program combining every concept: LabelFrames, Labels, Entries, Spinboxes, Comboboxes, Checkbuttons, and a Button that reads all values on click.

import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.geometry("500x460")
root.title("Data Entry Form")

# ── User Information ──────────────────────────
frame_info = tk.LabelFrame(root, text="User Information", padx=5, pady=5)
frame_info.place(x=10, y=5, width=470, height=145)

tk.Label(frame_info, text="First Name").place(x=50, y=5)
tk.Label(frame_info, text="Last Name").place(x=195, y=5)
tk.Label(frame_info, text="Title").place(x=360, y=5)

entry_fname = tk.Entry(frame_info, width=15); entry_fname.place(x=15, y=25)
entry_lname = tk.Entry(frame_info, width=15); entry_lname.place(x=160, y=25)

titles = ["Mr", "Ms", "Mrs", "Dr", "Prof"]
combo_title = ttk.Combobox(frame_info, values=titles, width=10)
combo_title.place(x=330, y=25)

tk.Label(frame_info, text="Age").place(x=50, y=65)
tk.Label(frame_info, text="Nationality").place(x=195, y=65)

spin_age = tk.Spinbox(frame_info, from_=16, to=99, width=12)
spin_age.place(x=15, y=85)
nations = ["British", "American", "French", "German", "Other"]
combo_nation = ttk.Combobox(frame_info, values=nations, width=13)
combo_nation.place(x=160, y=85)

# ── Registration Status ───────────────────────
frame_reg = tk.LabelFrame(root, text="Registration Status", padx=5, pady=5)
frame_reg.place(x=10, y=160, width=470, height=75)

tk.Label(frame_reg, text="# Completed Courses").place(x=160, y=5)
tk.Label(frame_reg, text="# Semesters").place(x=320, y=5)

var_reg = tk.BooleanVar()
tk.Checkbutton(frame_reg, text="Currently Registered",
             variable=var_reg).place(x=5, y=30)
spin_courses = tk.Spinbox(frame_reg, from_=0, to=50, width=10)
spin_courses.place(x=160, y=30)
spin_sem = tk.Spinbox(frame_reg, from_=0, to=20, width=10)
spin_sem.place(x=320, y=30)

# ── Terms & Conditions ───────────────────────
frame_terms = tk.LabelFrame(root, text="Terms & Conditions", padx=5, pady=5)
frame_terms.place(x=10, y=245, width=470, height=55)

var_terms = tk.BooleanVar()
tk.Checkbutton(frame_terms, text="I accept the terms and conditions.",
             variable=var_terms).place(x=5, y=5)

# ── Submit button ─────────────────────────────
def submit():
    data = {
        "first_name": entry_fname.get(),
        "last_name": entry_lname.get(),
        "title": combo_title.get(),
        "age": spin_age.get(),
        "nationality": combo_nation.get(),
        "registered": var_reg.get(),
        "courses": spin_courses.get(),
        "semesters": spin_sem.get(),
        "terms": var_terms.get(),
    }
    print(data)

tk.Button(root, text="Enter data", width=62,
        command=submit).place(x=10, y=415)

root.mainloop()

Window setup

import tkinter as tk
from tkinter import ttk
root = tk.Tk()
root.geometry("500x400")
root.title("My App")
# ... widgets ...
root.mainloop()

Label & Entry

# Label
tk.Label(root, text="Name").place(x=10, y=10)

# Entry (text field)
e = tk.Entry(root, width=20)
e.place(x=10, y=35)
value = e.get() # read text

Button

# Define function first
def my_func():
  print("clicked!")

# command= no brackets!
tk.Button(root,
  text="Click",
  command=my_func).pack()

Spinbox & Combobox

# Number picker
s = tk.Spinbox(root, from_=0, to=100)
s.place(x=10, y=10)

# Dropdown
opts = ["A", "B", "C"]
c = ttk.Combobox(root, values=opts)
c.place(x=10, y=40)
picked = c.get()

Checkbutton

# Must link to a BooleanVar
var = tk.BooleanVar()

tk.Checkbutton(root,
  text="Accept terms",
  variable=var).place(x=10, y=10)

is_ticked = var.get() # True/False

LabelFrame

# Grouped border box
f = tk.LabelFrame(root,
  text="Section",
  padx=5, pady=5)
f.place(x=10, y=10, width=300, height=100)

# Parent is f, not root!
tk.Label(f, text="Inside").place(x=5, y=5)

Placement methods

# .place() — exact pixels
widget.place(x=50, y=100)

# .pack() — stack top to bottom
widget.pack(pady=10)

# .grid() — row/column table
widget.grid(row=0, column=1, padx=5)

Reading values

# Entry, Spinbox, Combobox
text = entry.get()
num = spinbox.get() # string!
sel = combobox.get()

# BooleanVar (Checkbutton)
is_on = bool_var.get() # True/False

# Convert number string to int
age = int(spinbox.get())
