What is tkinter?

Python's built-in GUI library

tkinter comes installed with Python — no pip install needed. It lets you build real desktop windows with buttons, inputs, dropdowns, and more.

🐍

Built into Python

tkinter ships with every Python installation. Just import tkinter as tk and you're ready — no packages to install.

🪟

Cross-platform

Write once, run on Windows, macOS, and Linux. Your app's window adapts to the native look of each operating system automatically.

🧩

Widget-based

Everything is a widget — labels, buttons, text boxes, dropdowns. You place them on a window (called the "root") and arrange them using pack, grid, or place.

⚡

Event-driven

tkinter uses an event loop — root.mainloop(). The window waits for the user to interact, then runs your functions in response.

Widget library

Click any widget below to see its properties and a ready-to-copy code example.

tk.Label

Displays text or an image on screen. Labels are read-only — users can't edit them. Use them for titles, field names, and instructions.

  • text=The string to display
  • font=("Arial", 12) — family and size
  • fg= / bg=Foreground / background colour
  • width=Width in characters
  • anchor=Text alignment: "w", "center", "e"
import tkinter as tk

root = tk.Tk()
root.geometry("300x150")

# Simple label — .pack() centres it
tk.Label(root, text="Hello, class!",
       font=("Arial", 16),
       fg="#2ecc71").pack(pady=20)

# Positioned label
tk.Label(root, text="First Name").place(x=20, y=80)

root.mainloop()

tk.Entry

A single-line text input box. Save it in a variable so you can read what the user typed using .get().

  • width=Width in characters (default 20)
  • show=Use "*" for password fields
  • state="normal", "disabled", "readonly"
  • .get()Returns current text as a string
  • .delete(0,END)Clears the field
import tkinter as tk

root = tk.Tk()
root.geometry("300x150")

tk.Label(root, text="First Name").place(x=20, y=20)

# Save entry in a variable to read later
entry_fname = tk.Entry(root, width=20)
entry_fname.place(x=20, y=45)

def read_name():
    name = entry_fname.get()
    print(f"Hello {name}!")

tk.Button(root, text="Read", command=read_name).place(x=20, y=80)
root.mainloop()

tk.Button

A clickable button. Use command= to point to a function — but don't use brackets, or it runs immediately!

  • text=Label on the button
  • command=Function to call (no brackets!)
  • width=Width in characters
  • bg= / fg=Background / text colour
  • state="normal" or "disabled"
import tkinter as tk

root = tk.Tk()
root.geometry("300x120")

def say_hello():
    print("Button clicked!")

# command= takes the function NAME, no brackets
tk.Button(root,
        text="Click Me",
        width=20,
        command=say_hello).pack(pady=30)

root.mainloop()

tk.Spinbox

A number picker with up/down arrows. Set the minimum with from_= (underscore because from is a Python keyword) and maximum with to=.

  • from_=Minimum value (note the underscore)
  • to=Maximum value
  • width=Width in characters
  • increment=Step size (default 1)
  • .get()Returns current value as a string
import tkinter as tk

root = tk.Tk()
root.geometry("300x120")

tk.Label(root, text="Age").place(x=20, y=20)

# from_= has underscore — "from" is a Python keyword
spin_age = tk.Spinbox(root, from_=16, to=99, width=12)
spin_age.place(x=20, y=45)

def show_age():
    print(spin_age.get())

tk.Button(root, text="Print age", command=show_age).place(x=20, y=80)
root.mainloop()

ttk.Combobox

A dropdown menu from the ttk module (themed widgets). Pass a Python list to values= to populate the options.

  • values=A list of options to show
  • width=Width in characters
  • state="readonly" prevents typing
  • .get()Returns the selected option
  • .set("x")Sets the default value
import tkinter as tk
from tkinter import ttk # ttk has nicer widgets

root = tk.Tk()
root.geometry("300x120")

tk.Label(root, text="Title").place(x=20, y=20)

titles = ["Mr", "Ms", "Mrs", "Dr", "Prof"]
combo = ttk.Combobox(root, values=titles, width=12)
combo.place(x=20, y=45)

tk.Button(root, text="Print",
        command=lambda: print(combo.get())).place(x=20, y=80)
root.mainloop()

tk.Checkbutton

A tick box. It must be linked to a tk.BooleanVar() — the variable stores True when ticked, False when not.

  • text=Label next to the tick box
  • variable=A BooleanVar() to store the state
  • onvalue= / offvalue=Custom values (default True/False)
  • .get()Returns True or False
  • .set(True)Ticks the box programmatically
import tkinter as tk

root = tk.Tk()
root.geometry("300x140")

# BooleanVar holds the tick state
var_reg = tk.BooleanVar()
var_terms = tk.BooleanVar()

tk.Checkbutton(root, text="Currently Registered",
             variable=var_reg).place(x=20, y=30)

tk.Checkbutton(root, text="Accept terms",
             variable=var_terms).place(x=20, y=60)

tk.Button(root, text="Check",
        command=lambda: print(var_reg.get(), var_terms.get())
        ).place(x=20, y=100)
root.mainloop()

tk.LabelFrame

A bordered box with a title, used to group related widgets. Widgets placed inside it use the frame as their parent instead of root.

  • text=Title shown in the frame border
  • padx= / pady=Inner padding in pixels
  • width= / height=Size in pixels (used with .place())
  • relief="groove", "ridge", "flat", "sunken"
import tkinter as tk

root = tk.Tk()
root.geometry("350x200")

# LabelFrame — place at a position on root
frame = tk.LabelFrame(root, text="User Information",
                    padx=5, pady=5)
frame.place(x=10, y=10, width=330, height=120)

# Widgets inside use frame as parent, not root
tk.Label(frame, text="First Name").place(x=10, y=5)
entry = tk.Entry(frame, width=15)
entry.place(x=10, y=30)

root.mainloop()

Step 1 — Create the window

Every tkinter program has the same skeleton: import, create the window, set its size and title, then call mainloop() to keep it open.

import tkinter as tk
from tkinter import ttk

root = tk.Tk() # create the window
root.geometry("500x600") # width x height
root.title("IT Club Registry") # title bar text

# Always the last line — keeps the window open
root.mainloop()

Step 2 — Add labels with .place()

Labels display text. Use .place(x=, y=) to put them at exact pixel coordinates.

# ... (root setup) ...

# Labels — .place(x, y) sets exact pixel position

tk.Label(root, text="Last Name").place(x=180, y=30)
tk.Label(root, text="Middle Name").place(x=330, y=30)

tk.Label(root, text="Essay (Why join?)").place(x=30, y=150)

root.mainloop()

Step 7 — The Final Code

Here is the completed code for the IT Club Registry, organized neatly into frames using .grid() and .pack() for a cleaner look.

import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.geometry("500x600")
root.title("IT Club Registry")

# ── Personal Details ─────────────────────────
frame_p = tk.LabelFrame(root, text="Personal Details", padx=10, pady=10)
frame_p.pack(padx=20, pady=10, fill="x")

tk.Label(frame_p, text="First Name").grid(row=0, column=0, sticky="w")
entry_f = tk.Entry(frame_p, width=15)
entry_f.grid(row=1, column=0, padx=5)

tk.Label(frame_p, text="Last Name").grid(row=0, column=1, sticky="w")
entry_l = tk.Entry(frame_p, width=15)
entry_l.grid(row=1, column=1, padx=5)

tk.Label(frame_p, text="Middle Name").grid(row=0, column=2, sticky="w")
entry_m = tk.Entry(frame_p, width=15)
entry_m.grid(row=1, column=2, padx=5)

# ── Academic Details ─────────────────────────
frame_a = tk.LabelFrame(root, text="Academic & Gender", padx=10, pady=10)
frame_a.pack(padx=20, pady=10, fill="x")

tk.Label(frame_a, text="Grade").grid(row=0, column=0, sticky="w")
combo_g = ttk.Combobox(frame_a, values=("9", "10", "11", "12"), width=8)
combo_g.grid(row=1, column=0, padx=5)

tk.Label(frame_a, text="Section").grid(row=0, column=1, sticky="w")
combo_s = ttk.Combobox(frame_a, values=("A", "B", "C", "D"), width=8)
combo_s.grid(row=1, column=1, padx=5)

tk.Label(frame_a, text="Gender").grid(row=0, column=2, sticky="w", padx=15)
gender_var = tk.StringVar(value="Male")
tk.Radiobutton(frame_a, text="M", variable=gender_var, value="Male").grid(row=1, column=2, sticky="w", padx=(15,0))
tk.Radiobutton(frame_a, text="F", variable=gender_var, value="Female").grid(row=1, column=2, sticky="e")

# ── Essay ────────────────────────────────────
frame_e = tk.LabelFrame(root, text="Why do you want to join the IT Club?", padx=10, pady=10)
frame_e.pack(padx=20, pady=10, fill="x")

text_essay = tk.Text(frame_e, height=4, width=50)
text_essay.pack()

# ── Agreement ────────────────────────────────
frame_t = tk.LabelFrame(root, text="Agreement", padx=10, pady=5)
frame_t.pack(padx=20, pady=5, fill="x")

terms_var = tk.BooleanVar()
tk.Checkbutton(frame_t, text="I agree to the terms of the IT club.", variable=terms_var).pack(anchor="w")

# ── Submit Button ────────────────────────────
btn_submit = tk.Button(root, text="Register Student", bg="#4fffb0", fg="black")
btn_submit.pack(pady=15)

root.mainloop()

1. Window & Display

We need a window and a large entry box at the top to display numbers.

import tkinter as tk

root = tk.Tk()
root.title("Calculator")
root.geometry("300x400")

# Entry for display
display = tk.Entry(root, font=("Arial", 24), justify="right")
display.grid(row=0, column=0, columnspan=4, sticky="we", pady=10, padx=10)

root.mainloop()

2. Number Buttons

Let's add buttons for numbers 7, 8, 9 using the .grid() manager.

# Buttons Row 1
tk.Button(root, text="7", width=5, height=2).grid(row=1, column=0, padx=5, pady=5)

3. Click Logic

Now, let's create a function that runs when a button is clicked, inserting the clicked number into the display.

def button_click(number):
    current = display.get()
    display.delete(0, tk.END)
    display.insert(0, str(current) + str(number))

# Use lambda to pass arguments to the command function
tk.Button(root, text="7", command=lambda: button_click(7)).grid(row=1, column=0)

4. Eval & Clear

Finally, we need a Clear button and an Equals button to calculate the result using Python's eval() function.

def button_clear():
    display.delete(0, tk.END)

def button_equal():
    current = display.get()
    display.delete(0, tk.END)
    try:
        result = str(eval(current))
        display.insert(0, result)
    except Exception:
        display.insert(0, "Error")

# Add equal and clear buttons
tk.Button(root, text="C", command=button_clear, bg="#ff6b6b").grid(row=5, column=0, columnspan=2, sticky="we")
tk.Button(root, text="=", command=button_equal, bg="#4fffb0").grid(row=5, column=2, columnspan=2, sticky="we")

5. Complete Code

Here is the full calculator code put together. Copy and run this in your Python environment!

import tkinter as tk

def button_click(number):
    current = display.get()
    display.delete(0, tk.END)
    display.insert(0, str(current) + str(number))

def button_clear():
    display.delete(0, tk.END)

def button_equal():
    current = display.get()
    display.delete(0, tk.END)
    try:
        result = str(eval(current))
        display.insert(0, result)
    except Exception:
        display.insert(0, "Error")

root = tk.Tk()
root.title("Calculator")
root.geometry("300x400")
